export default { async fetch(request, env) { const corsHeaders = { "Access-Control-Allow-Origin": "*", "Access-Control-Allow-Methods": "GET, POST, OPTIONS", "Access-Control-Allow-Headers": "Content-Type, Authorization", }; if (request.method === "OPTIONS") return new Response(null, { headers: corsHeaders }); const url = new URL(request.url); const method = request.method; try { // 1. 登录 if (url.pathname === "/api/login" && method === "POST") { const { userId, password } = await request.json(); const user = await env.DB.prepare("SELECT * FROM users WHERE userId = ? AND password = ?").bind(userId, password).first(); return new Response(JSON.stringify({ success: !!user, error: user ? null : "账号或密码错误" }), { headers: corsHeaders }); } // 2. 注册 (已增加查重与友好提示) if (url.pathname === "/api/register" && method === "POST") { const { userId, password } = await request.json(); const existing = await env.DB.prepare("SELECT userId FROM users WHERE userId = ?").bind(userId).first(); if (existing) { return new Response(JSON.stringify({ success: false, error: "已有人注册此账号" }), { headers: corsHeaders }); } await env.DB.prepare("INSERT INTO users (userId, password) VALUES (?, ?)").bind(userId, password).run(); return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } // 3. 获取列表 if (url.pathname === "/api/list") { const userId = url.searchParams.get("userId"); const friends = await env.DB.prepare("SELECT friendId FROM friends WHERE userId = ?").bind(userId).all(); const groups = await env.DB.prepare("SELECT groupId FROM groups WHERE userId = ?").bind(userId).all(); return new Response(JSON.stringify({ friends: friends ? friends.results.map(f => f.friendId) : [], groups: groups ? groups.results.map(g => g.groupId) : [] }), { headers: corsHeaders }); } // 4. 获取聊天记录 (双向私聊与群聊查询) if (url.pathname === "/api/messages") { const userId = url.searchParams.get("userId"); const targetId = url.searchParams.get("targetId"); const type = url.searchParams.get("type"); let msgs; if (type === "group") { msgs = await env.DB.prepare("SELECT * FROM messages WHERE to_id = ? AND type='group' ORDER BY timestamp ASC").bind(targetId).all(); } else { msgs = await env.DB.prepare("SELECT * FROM messages WHERE (from_user = ? AND to_id = ? AND type='private') OR (from_user = ? AND to_id = ? AND type='private') ORDER BY timestamp ASC").bind(userId, targetId, targetId, userId).all(); } return new Response(JSON.stringify(msgs ? msgs.results : []), { headers: corsHeaders }); } // 5. 发送消息 if (url.pathname === "/api/send" && method === "POST") { const { from, to, type, text } = await request.json(); await env.DB.prepare("INSERT INTO messages (from_user, to_id, type, content) VALUES (?, ?, ?, ?)").bind(from, to, type, text).run(); return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } // 6. 加好友 (双向写入) if (url.pathname === "/api/add-friend" && method === "POST") { const { userId, friendId } = await request.json(); await env.DB.prepare("INSERT OR IGNORE INTO friends (userId, friendId) VALUES (?, ?)").bind(userId, friendId).run(); await env.DB.prepare("INSERT OR IGNORE INTO friends (userId, friendId) VALUES (?, ?)").bind(friendId, userId).run(); return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } // 7. 进群 if (url.pathname === "/api/join-group" && method === "POST") { const { userId, groupId } = await request.json(); await env.DB.prepare("INSERT OR IGNORE INTO groups (userId, groupId) VALUES (?, ?)").bind(userId, groupId).run(); return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } // 8. 删除好友 (双向删除) if (url.pathname === "/api/delete-friend" && method === "POST") { const { userId, friendId } = await request.json(); await env.DB.prepare("DELETE FROM friends WHERE (userId = ? AND friendId = ?) OR (userId = ? AND friendId = ?)").bind(userId, friendId, friendId, userId).run(); return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } // 9. 退出群聊 if (url.pathname === "/api/leave-group" && method === "POST") { const { userId, groupId } = await request.json(); await env.DB.prepare("DELETE FROM groups WHERE userId = ? AND groupId = ?").bind(userId, groupId).run(); return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } // 10. 管理后台：获取全量数据 if (url.pathname === "/api/admin/data") { const users = await env.DB.prepare("SELECT userId FROM users").all(); const groups = await env.DB.prepare("SELECT DISTINCT groupId FROM groups").all(); const messages = await env.DB.prepare("SELECT from_user, to_id, content, timestamp FROM messages ORDER BY timestamp DESC LIMIT 50").all(); return new Response(JSON.stringify({ users: users ? users.results : [], groups: groups ? groups.results : [], messages: messages ? messages.results : [] }), { headers: corsHeaders }); } // 11. 管理后台：删除用户或群聊 if (url.pathname === "/api/admin/delete" && method === "POST") { const { type, id } = await request.json(); if (type === "user") { await env.DB.prepare("DELETE FROM users WHERE userId = ?").bind(id).run(); await env.DB.prepare("DELETE FROM friends WHERE userId = ? OR friendId = ?").bind(id, id).run(); await env.DB.prepare("DELETE FROM groups WHERE userId = ?").bind(id).run(); } else if (type === "group") { await env.DB.prepare("DELETE FROM groups WHERE groupId = ?").bind(id).run(); } return new Response(JSON.stringify({ success: true }), { headers: corsHeaders }); } return new Response("Not Found", { status: 404, headers: corsHeaders }); } catch (e) { return new Response(JSON.stringify({ error: e.message }), { status: 500, headers: corsHeaders }); } } };
